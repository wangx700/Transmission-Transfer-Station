0014: """Megatron-side delta export machinery, built on Megatron-Bridge param mappings.
0015: 
0016: The delta engine consumes final HF-coordinate entries; everything mcore-specific
0017: lives here: enumerating parameters through :meth:`AutoBridge.get_conversion_tasks`,
0018: routing each parameter's entries to its wire merge group, and probing the
0019: bridge's own ``megatron_to_hf`` converters with NaN sentinels to translate a
0020: shard-local delta into HF coordinates.
0021: 
0022: The probe never runs a collective, yet executes the mapping's REAL parallel
0023: code paths: a "group" carries two separable meanings -- its SIZE/RANK (which
0024: value math like Mamba's ``local_dim = global // tp_size`` consumes) and its
0025: COMMUNICATION. The probe copies keep the first faithful and replace only the
0026: second: groups become :class:`_ProbeGroup` (real size/rank, any actual use for
0027: communication raises), and the bridge's comm helpers are stubbed with local
0028: synthesis -- ``gather_from_tp_ranks`` returns this rank's shard at its true
0029: rank index with NaN placeholders for every other rank (their contributions are
0030: exported by those ranks' own probes). Feeding ``megatron_to_hf`` the LOCAL
0031: shard as a NaN buffer with the rank's own delta scattered in yields HF tensors
0032: whose non-NaN survivors are exactly this rank's contributions in final HF
0033: coordinates.
0034: 
0035: Scope (asserted in the exporter): TP + EP + PP/VPP, no LoRA. Under PP the
0036: bridge's conversion tasks already enumerate the GLOBAL parameter directory
0037: (identical order on every rank, tied embeddings deduped, placeholders for
0038: other stages); non-owner ranks ship zero-count lockstep rows and every
0039: param merges over the WORLD group, so the wire master needs no relay.
0040: 
0041: Safety boundary: communication must stay confined to the four stubbed helpers
0042: (``gather_from_tp_ranks`` / ``gather_from_ep_ranks[_scale]`` / the PP
0043: broadcasts, whose ``pp_size == 1`` fast path the probe keeps) -- anything else
0044: touching a probe group raises instead of skewing silently. The remaining
0045: assumption is that transforms REARRANGE elements rather than arithmetically
0046: BLEND chunks (blending would eat the NaN sentinels); the TP>1 differential
0047: test (real ``megatron_to_hf`` vs probe assembly, bitwise) is the regression
0048: oracle for both assumptions on every Megatron-Bridge upgrade.

0091: def _nan_block(shape, dtype, device, pool) -> torch.Tensor:
0092:     """Read-only all-NaN placeholder for another rank's gather chunk, pooled by
0093:     (shape, dtype, device): the gather stub hands these out on every probe call
0094:     for every param, and they are only ever READ (the transforms are functional),
0095:     so one block per distinct shape serves the whole model for the run's
0096:     lifetime instead of a fresh cudaMalloc+fill per param per sync."""

0098:     t = pool.get(key)
0099:     if t is None:
0100:         t = torch.full(tuple(shape), float("nan"), dtype=dtype, device=device)
0101:         pool[key] = t

0137:     def _stub(c):
0138:         # size-faithful groups, sizes/ranks read from the ORIGINAL (real) groups
0139:         # BEFORE replacement. The probe only ever runs on the stage that OWNS
0140:         # the param, where a pp broadcast is the identity -- a (1, 0) group
0141:         # keeps the pp helpers on that fast path regardless of the real pp size.
0142:         c.pp_group = _ProbeGroup(1, 0)
0143:         for attr in ("ep_group", "_tp_group", "_etp_group"):
0144:             g = getattr(c, attr, None)
0145:             setattr(c, attr, _ProbeGroup(get_pg_size(g), get_pg_rank(g)))
0146: 
0147:         def _gather_tp(tensor, _c=c):
0148:             # what a real all_gather over the tp group would produce, minus the
0149:             # other ranks' data: our shard rides at its true rank index, every
0150:             # other chunk is a pooled read-only NaN block (those ranks export
0151:             # their own contributions; transforms never write into their inputs,
0152:             # which the TP>1 bitwise differential revalidates).
0153:             nan = _nan_block(tensor.shape, tensor.dtype, tensor.device, nan_pool)
0154:             out = [nan] * _c.tp_size
0155:             out[_c.tp_rank] = tensor
0156:             return out

0158:         c.gather_from_tp_ranks = _gather_tp
0159:         # each rank's probe emits only its own local experts, under the hf name
0160:         # bound with the GLOBAL expert id at task construction; the engine
0161:         # merges entries over the etp x ep group, so the ep fan-out reduces to
0162:         # the bound name.
0163:         c.gather_from_ep_ranks = lambda w, mod, name: {str(name): w}
0164:         # mirrors the real helper's tail (unsqueeze(0) ... squeeze().unsqueeze(-1))
0165:         c.gather_from_ep_ranks_scale = lambda w, mod, name: {str(name): w.unsqueeze(0).squeeze().unsqueeze(-1)}
0166:         return c

0168:     def _inject(m):
0169:         c = _stub(copy.copy(m))
0170:         for attr, value in list(vars(c).items()):
0171:             if isinstance(value, MegatronParamMapping):
0172:                 # warm BEFORE copying the child: the lazy delegate must exist so
0173:                 # the child copy's vars() include it and the recursion reaches it.
0174:                 _warm_lazy_mappings(value, module)
0175:                 setattr(c, attr, _inject(value))
0176:         return c

0392: def _direct_hf_delta_entry(rec, lidx, lval, *, sorted_indices=False, midpoint=None):
0393:     """Sparse equivalents of exact Bridge mappings; unknown transforms use the probe.
0394: 
0395:     Use Bridge's resolved names and TP/ETP properties, never model-name guesses.
0396:     Subclasses may implement arithmetic transforms and deliberately do not match.
0397:     """
0398:     from megatron.bridge.models.conversion.param_mapping import (
0399:         AutoMapping,
0400:         ColumnParallelMapping,
0401:         GatedMLPMapping,
0402:         ReplicatedMapping,

0412:         return None

0414:         return None

0417:     pieces = {}
0418:     if type(mapping) is GatedMLPMapping:
0419:         if shape[0] % 2:
0420:             return None

0422:         out_shape = (shape[0] // 2 * size, *shape[1:])
0423:         if sorted_indices:
0424:             # nonzero emits ascending shard positions. One scalar read replaces
0425:             # four dynamic boolean selections; generic callers retain fallback.

0427:         for part, key in enumerate(("gate", "up")):

0429:                 (slice(0, split) if part == 0 else slice(split, None))

0432:             )

0458:         ):
0459:             return None

0464:             continue
0465:         _, indices, values = pieces[name]
0466:         if tuple(shape) != pieces[name][0]:
0467:             limit = 1
0468:             for dim in shape:
0469:                 limit *= dim
0470:             selected = indices < limit
0471:             indices, values = indices[selected], values[selected]
0472:         counts[slot] = indices.numel()

0474:         value_parts.append(values)

0478: def mcore_hf_delta_entry(
0479:     rec: McoreParamExport,
0480:     _place,
0481:     lidx: torch.Tensor,
0482:     lval: torch.Tensor,
0483:     slot_cache: dict,
0484:     *,
0485:     force_probe=False,
0486:     sorted_indices=False,
0487:     midpoint=None,
0488: ):
0489:     """Probe one mcore param's shard-local delta into its final HF-coordinate
0490:     entry ``(slots, dtype_str, counts, hf_idx, hf_val)``.
0491: 
0492:     Scatters the delta into a NaN buffer of the LOCAL shard shape (exactly what
0493:     the real ``megatron_to_hf`` receives), runs the comm-stubbed probe -- real
0494:     group sizes, gathers synthesized locally, so the mapping executes its real
0495:     TP>1 code paths -- and extracts each output slot's surviving positions.
0496:     The slot list is cached after the first call (the converter's output names
0497:     are deterministic, so every rank's cache agrees and the batched gather
0498:     stays aligned)."""
0499:     dtype_str = str(lval.dtype).replace("torch.", "")
0500:     assert (
0501:         lval.numel() == 0 or lval.is_floating_point()
0502:     ), f"{rec.megatron_name}: NaN sentinels require a floating-point param, got {lval.dtype}"
0503: 
0504:     cached = rec.slots if rec.slots is not None else slot_cache.get(rec.megatron_name)
0505:     if rec.param is None:
0506:         # owned by another pipeline stage: pure lockstep row. The slot table
0507:         # was pre-seeded from the owner stage (fail loud if not -- an
0508:         # unsized entry would silently misalign the batched gather).
0509:         assert cached is not None, f"{rec.megatron_name}: slot table not pre-seeded for a non-owned PP param"
0510:         assert lidx.numel() == 0, f"{rec.megatron_name}: delta reported for a param this rank does not own"
0511:     if lidx.numel() == 0 and cached is not None:
0512:         # empty delta: the slot table froze after the first probe, so the
0513:         # zero-count lockstep entry needs no probe run at all -- skip the
0514:         # buffer build, the transform and the full-output NaN scan.
0515:         return (
0516:             cached,
0517:             dtype_str,
0518:             torch.zeros(len(cached), dtype=torch.int64),
0519:             torch.empty(0, dtype=torch.int32, device=lval.device),
0520:             torch.empty(0, dtype=lval.dtype, device=lval.device),
0521:         )
0522: 
0523:     if not force_probe and rec.slots is not None:
0524:         direct = _direct_hf_delta_entry(rec, lidx, lval, sorted_indices=sorted_indices, midpoint=midpoint)
0525:         if direct is not None:
0526:             return direct
0527: 
0528:     buf = torch.full(tuple(rec.param.shape), float("nan"), dtype=lval.dtype, device=lval.device)
0529:     if lidx.numel():
0530:         buf.view(-1)[lidx] = lval
0531: 
0532:     outs = {
0533:         name: trim_hf_vocab_padding(name, tensor, rec.hf_vocab_size)
0534:         for name, tensor in rec.probe.megatron_to_hf(buf, rec.module).items()
0535:     }
0536: 
0537:     key = rec.megatron_name
0538:     slots = rec.slots if rec.slots is not None else slot_cache.get(key)
0539:     if slots is None:
0540:         # only reachable from the index-build exchange itself, which probes a
0541:         # zero delta to reveal this rank's local list before the row unions
0542:         # are installed on the recs; steady entries always find rec.slots.
0543:         slots = [(n, tuple(int(x) for x in t.shape)) for n, t in outs.items()]
0544:         slot_cache[key] = slots
0545:     unknown = set(outs) - {n for n, _ in slots}
0546:     assert not unknown, (
0547:         f"{key}: probe emitted slots missing from the global table {sorted(unknown)[:4]} -- "
0548:         "non-deterministic converter naming would misalign the batched gather"
0549:     )
0550:     counts = torch.zeros(len(slots), dtype=torch.int64)
0551:     idx_pieces: list[torch.Tensor] = []
0552:     val_pieces: list[torch.Tensor] = []
0553:     for s_i, (sname, _sshape) in enumerate(slots):
0554:         out = outs.get(sname)
0555:         if out is None:
0556:             continue  # another rank's slot (e.g. its ep-local experts): zero count here
0557:         fl = out.reshape(-1)
0558:         p_ = (~torch.isnan(fl)).nonzero(as_tuple=False).view(-1)
0559:         if p_.numel():
0560:             counts[s_i] = p_.numel()
0561:             idx_pieces.append(p_.to(torch.int32))
0562:             val_pieces.append(fl[p_])
0563:     if idx_pieces:
0564:         hf_idx = torch.cat(idx_pieces)
0565:         hf_val = torch.cat(val_pieces)
0566:     else:
0567:         hf_idx = torch.empty(0, dtype=torch.int32, device=lval.device)
0568:         hf_val = torch.empty(0, dtype=lval.dtype, device=lval.device)
0569:     return slots, dtype_str, counts, hf_idx, hf_val

0593: def validate_export_weights(index):
0594:     """Reject NaN on every rank before payloads or snapshot changes.
0595: 
0596:     NaN is the probe/loader sentinel. Reduce device-side flags once per update,
0597:     before ranks can enter different payload collectives; a local exception
0598:     during streaming would strand peers. No per-parameter host synchronization.
0599:     """
0600:     flags = [torch.isnan(rec.param.detach()).any() for rec in index if rec.param is not None]
0601:     invalid = (
0602:         torch.stack(flags).any().to(torch.int32)
0603:         if flags
0604:         else torch.zeros((), dtype=torch.int32, device=torch.accelerator.current_device_index())
0605:     )
0606:     torch.distributed.all_reduce(invalid, op=torch.distributed.ReduceOp.MAX)
0607:     if bool(invalid):
0608:         raise ValueError("Sparse HCCL cannot export NaN weights; update rejected on all trainer ranks")

0611: class _SnapshotRefreshQueue:
0612:     """Bound sparse baseline refreshes by bytes and row count.
0613: 
0614:     Two blocking packed copies per batch establish host visibility before
0615:     index_copy_. This avoids two device synchronizations for every parameter.
0616:     The queue is drained before the generator completes and the next version.
0617:     """
0618: 
0619:     def __init__(self, max_bytes=32 << 20, max_rows=32):
0620:         self.max_bytes, self.max_rows = max_bytes, max_rows
0621:         self.rows = []
0622:         self.nbytes = 0
0623: 
0624:     def add(self, snapshot, indices, values):
0625:         if not indices.numel():
0626:             return
0627:         size = indices.numel() * (indices.element_size() + values.element_size())
0628:         if self.rows and (self.nbytes + size > self.max_bytes or self.rows[0][2].dtype != values.dtype):
0629:             self.flush()
0630:         self.rows.append((snapshot, indices, values))
0631:         self.nbytes += size
0632:         if self.nbytes >= self.max_bytes or len(self.rows) >= self.max_rows:
0633:             self.flush()
0634: 
0635:     def flush(self):
0636:         if not self.rows:
0637:             return
0638:         indices = torch.cat([row[1] for row in self.rows]).to("cpu", non_blocking=False)
0639:         values = torch.cat([row[2] for row in self.rows]).to("cpu", non_blocking=False)
0640:         offset = 0
0641:         for snapshot, local_indices, _ in self.rows:
0642:             end = offset + local_indices.numel()
0643:             snapshot.index_copy_(0, indices[offset:end], values[offset:end])
0644:             offset = end
0645:         self.rows.clear()
0646:         self.nbytes = 0

0649: def _iter_delta_entries_unbatched(index, snapshots):
0650:     """The verl hf_delta_export contract, with explicit Megatron placement."""
0651:     # CPU snapshots are required when a large training shard cannot retain a
0652:     # second model-sized copy on the accelerator.  Reuse one accelerator
0653:     # staging allocation per dtype/device instead of allocating a new Tensor
0654:     # for every parameter via snapshot.to(device).  Pinned snapshots make the
0655:     # copy asynchronous; current-stream ordering keeps it complete before the
0656:     # NPU comparison below and before the buffer is reused by the next row.
0657:     staging: dict[tuple[torch.device, torch.dtype], torch.Tensor] = {}
0658:     refresh = _SnapshotRefreshQueue()
0659:     for record in index:
0660:         if record.param is None:
0661:             device = torch.accelerator.current_device_index()
0662:             local_indices = torch.empty(0, dtype=torch.int64, device=device)
0663:             local_values = torch.empty(0, dtype=record.dtype, device=device)
0664:         else:
0665:             local = _device_shard(record)
0666:             if record.spec.contributes:
0667:                 snapshot = snapshots.get(record.megatron_name)
0668:                 if snapshot is None or snapshot.shape != local.shape or snapshot.dtype != local.dtype:
0669:                     raise RuntimeError(f"{record.megatron_name}: no matching seed snapshot for this shard")
0670:                 if snapshot.device == local.device:
0671:                     base = snapshot
0672:                 else:
0673:                     key = (local.device, local.dtype)
0674:                     buffer = staging.get(key)
0675:                     if buffer is None or buffer.numel() < local.numel():
0676:                         buffer = torch.empty_like(local)
0677:                         staging[key] = buffer
0678:                     base = buffer[: local.numel()]
0679:                     base.copy_(snapshot, non_blocking=True)
0680:                 local_indices, local_values = shard_delta_indices(local, base, 0)
0681:                 if snapshot.device == local.device:
0682:                     snapshot.copy_(local, non_blocking=True)
0683:                 elif local_indices.numel():
0684:                     # Preserve the next baseline from the sparse result itself.
0685:                     # Copying only changed positions avoids a full model-shard
0686:                     # D2H transfer after every steady-state diff.
0687:                     refresh.add(snapshot, local_indices, local_values)
0688:             else:
0689:                 local_indices = torch.empty(0, dtype=torch.int64, device=local.device)
0690:                 local_values = torch.empty(0, dtype=local.dtype, device=local.device)
0691:         yield (
0692:             *mcore_hf_delta_entry(record, 0, local_indices, local_values, {}, sorted_indices=True),
0693:             record.spec.gather_group,
0694:         )
0695:     refresh.flush()

0698: def iter_delta_entries(index, snapshots, *, batch_size=32):
0699:     """Batch dynamic diff extraction, preserving verl's record/slot ordering."""
0700:     if batch_size <= 1:
0701:         yield from _iter_delta_entries_unbatched(index, snapshots)
0702:         return
0703:     from .delta_sync.sparse_gather import compact_shard_masks, shard_delta_mask
0704: 
0705:     max_mask_bytes = 32 << 20
0706:     pending, masks, locals_ = [], [], []
0707:     staging = {}
0708:     refresh = _SnapshotRefreshQueue()
0709:     mask_bytes = 0
0710:     batch_key = None
0711: 
0712:     def flush():
0713:         deltas = iter(compact_shard_masks(locals_, masks, with_midpoints=True))
0714:         for record, local, snapshot in pending:
0715:             midpoint = None
0716:             if snapshot is not None:
0717:                 indices, values, midpoint = next(deltas)
0718:                 if snapshot.device == local.device:
0719:                     snapshot.copy_(local, non_blocking=True)
0720:                 elif indices.numel():
0721:                     refresh.add(snapshot, indices, values)
0722:             else:
0723:                 device = local.device if local is not None else torch.accelerator.current_device_index()
0724:                 dtype = local.dtype if local is not None else record.dtype
0725:                 indices = torch.empty(0, dtype=torch.int64, device=device)
0726:                 values = torch.empty(0, dtype=dtype, device=device)
0727:             yield (
0728:                 *mcore_hf_delta_entry(record, 0, indices, values, {}, sorted_indices=True, midpoint=midpoint),
0729:                 record.spec.gather_group,
0730:             )
0731:         pending.clear()
0732:         masks.clear()
0733:         locals_.clear()
0734: 
0735:     for record in index:
0736:         local = _device_shard(record) if record.param is not None else None
0737:         active = local is not None and record.spec.contributes
0738:         key = (local.device, local.dtype) if active else batch_key
0739:         size = local.numel() if active else 0
0740:         if pending and (key != batch_key or mask_bytes + size > max_mask_bytes or len(pending) >= batch_size):
0741:             yield from flush()
0742:             mask_bytes = 0
0743:         batch_key = key
0744:         snapshot = None
0745:         if active:
0746:             snapshot = snapshots.get(record.megatron_name)
0747:             if snapshot is None or snapshot.shape != local.shape or snapshot.dtype != local.dtype:
0748:                 raise RuntimeError(f"{record.megatron_name}: no matching seed snapshot for this shard")
0749:             base = snapshot
0750:             if snapshot.device != local.device:
0751:                 buffer = staging.get(key)
0752:                 if buffer is None or buffer.numel() < local.numel():
0753:                     buffer = torch.empty_like(local)
0754:                     staging[key] = buffer
0755:                 base = buffer[: local.numel()]
0756:                 base.copy_(snapshot, non_blocking=True)
0757:             masks.append(shard_delta_mask(local, base))
0758:             locals_.append(local)
0759:             mask_bytes += size
0760:         pending.append((record, local, snapshot))
0761:     if pending:
0762:         yield from flush()
0763:     refresh.flush()
