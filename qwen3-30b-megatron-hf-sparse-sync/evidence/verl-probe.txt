0014: """Megatron-side delta export machinery, built on Megatron-Bridge param mappings.
0015: 
0016: The delta engine consumes final HF-coordinate entries; everything mcore-specific
0017: lives here: enumerating parameters through :meth:`AutoBridge.get_conversion_tasks`,
0018: routing each parameter's entries to its wire merge group, and probing the
0019: bridge's own ``megatron_to_hf`` converters with NaN sentinels to translate a
0020: shard-local delta into HF coordinates.
0021: 
0022: The probe never runs a collective, yet executes the mapping's REAL parallel
0023: code paths: a "group" carries two separable meanings -- its SIZE/RANK (which
0024: value math like Mamba's ``local_dim = global // tp_size`` consumes) and its
0025: COMMUNICATION. The probe copies keep the first faithful and replace only the
0026: second: groups become :class:`_ProbeGroup` (real size/rank, any actual use for
0027: communication raises), and the bridge's comm helpers are stubbed with local
0028: synthesis -- ``gather_from_tp_ranks`` returns this rank's shard at its true
0029: rank index with NaN placeholders for every other rank (their contributions are
0030: exported by those ranks' own probes). Feeding ``megatron_to_hf`` the LOCAL
0031: shard as a NaN buffer with the rank's own delta scattered in yields HF tensors
0032: whose non-NaN survivors are exactly this rank's contributions in final HF
0033: coordinates.
0034: 
0035: Scope (asserted in the exporter): TP + EP + PP/VPP, no LoRA. Under PP the
0036: bridge's conversion tasks already enumerate the GLOBAL parameter directory
0037: (identical order on every rank, tied embeddings deduped, placeholders for
0038: other stages); non-owner ranks ship zero-count lockstep rows and every
0039: param merges over the WORLD group, so the wire master needs no relay.
0040: 
0041: Safety boundary: communication must stay confined to the four stubbed helpers
0042: (``gather_from_tp_ranks`` / ``gather_from_ep_ranks[_scale]`` / the PP
0043: broadcasts, whose ``pp_size == 1`` fast path the probe keeps) -- anything else
0044: touching a probe group raises instead of skewing silently. The remaining
0045: assumption is that transforms REARRANGE elements rather than arithmetically
0046: BLEND chunks (blending would eat the NaN sentinels); the TP>1 differential
0047: test (real ``megatron_to_hf`` vs probe assembly, bitwise) is the regression
0048: oracle for both assumptions on every Megatron-Bridge upgrade.
0049: """

0094:     """Read-only all-NaN placeholder for another rank's gather chunk, pooled by

0124: def make_probe(mapping, module):
0125:     """Copy a Megatron-Bridge param mapping tree and turn ``megatron_to_hf``
0126:     into a communication-free LOCAL transform that still runs the REAL
0127:     (tp_size > 1) code paths: every copy's groups become size-faithful
0128:     :class:`_ProbeGroup` stand-ins and the bridge's comm helpers are stubbed
0129:     with local synthesis. The copy is recursive (composite mappings delegate to
0130:     inner mappings -- QKVMapping._tp_mapping is an AutoMapping which itself
0131:     delegates to a lazily-created concrete mapping; none of these receive the
0132:     outer stubbing on their own)."""
0133:     from megatron.bridge.models.conversion.param_mapping import MegatronParamMapping
0134:     from megatron.core.utils import get_pg_rank, get_pg_size
0135: 
0136:     _warm_lazy_mappings(mapping, module)

0138:     def _stub(c):
0139:         # size-faithful groups, sizes/ranks read from the ORIGINAL (real) groups
0140:         # BEFORE replacement. The probe only ever runs on the stage that OWNS
0141:         # the param, where a pp broadcast is the identity -- a (1, 0) group
0142:         # keeps the pp helpers on that fast path regardless of the real pp size.
0143:         c.pp_group = _ProbeGroup(1, 0)
0144:         for attr in ("ep_group", "_tp_group", "_etp_group"):
0145:             g = getattr(c, attr, None)
0146:             setattr(c, attr, _ProbeGroup(get_pg_size(g), get_pg_rank(g)))
0147: 
0148:         def _gather_tp(tensor, _c=c):
0149:             # what a real all_gather over the tp group would produce, minus the
0150:             # other ranks' data: our shard rides at its true rank index, every
0151:             # other chunk is a pooled read-only NaN block (those ranks export
0152:             # their own contributions; transforms never write into their inputs,
0153:             # which the TP>1 bitwise differential revalidates).
0154:             nan = _nan_block(tensor.shape, tensor.dtype, tensor.device)
0155:             out = [nan] * _c.tp_size
0156:             out[_c.tp_rank] = tensor
0157:             return out
0158: 
0159:         c.gather_from_tp_ranks = _gather_tp
0160:         # each rank's probe emits only its own local experts, under the hf name
0161:         # bound with the GLOBAL expert id at task construction; the engine
0162:         # merges entries over the etp x ep group, so the ep fan-out reduces to
0163:         # the bound name.
0164:         c.gather_from_ep_ranks = lambda w, mod, name: {str(name): w}
0165:         # mirrors the real helper's tail (unsqueeze(0) ... squeeze().unsqueeze(-1))
0166:         c.gather_from_ep_ranks_scale = lambda w, mod, name: {str(name): w.unsqueeze(0).squeeze().unsqueeze(-1)}
0167:         return c
0168: 
0169:     def _inject(m):
0170:         c = _stub(copy.copy(m))
0171:         for attr, value in list(vars(c).items()):
0172:             if isinstance(value, MegatronParamMapping):
0173:                 # warm BEFORE copying the child: the lazy delegate must exist so
0174:                 # the child copy's vars() include it and the recursion reaches it.
0175:                 _warm_lazy_mappings(value, module)
0176:                 setattr(c, attr, _inject(value))
0177:         return c
0178: 
0179:     return _inject(mapping)

0200:     and precompute its probe + wire routing.
0201: 
0202:     No shard geometry is hand-computed here: the comm-stubbed probe emits final
0203:     HF coordinates straight from the local shard, so the engine only needs to
0204:     know WHICH group's entries to merge per parameter (the spec's
0205:     ``gather_group``; ``full_shape`` is the nominal local shape) and which
0206:     ranks contribute. The index is built once (parameter sets are static) and
0207:     reused by both the shard export and the delta entry hook.
0208: 
0209:     Lockstep: ``get_conversion_tasks`` already enumerates the GLOBAL parameter
0210:     list in an order identical on every rank (the bridge allgathers and sorts
0211:     names across pp ranks, dedupes tied embeddings, and leaves
0212:     ``param_weight=None`` placeholders for parameters owned by other pipeline
0213:     stages). Under PP>1 those placeholders become zero-count lockstep rows and
0214:     every param's entries merge over the WORLD group (owner-stage ranks
0215:     contribute, dp/cp replicas and other stages stay empty); their slot tables
0216:     are pre-seeded once via :func:`_preseed_slot_tables`. Under PP=1 the
0217:     routing is unchanged (tp / etp x ep subgroups, replicated rank-0 direct).
0218:     """

0220: 
0221:     pp_world = mpu.get_pipeline_model_parallel_world_size()
0222:     tp_group = mpu.get_tensor_model_parallel_group()
0223:     tp_world = torch.distributed.get_world_size(group=tp_group)
0224:     ep_size = mpu.get_expert_model_parallel_world_size()
0225:     world = torch.distributed.group.WORLD
0226: 
0227:     tasks = bridge.get_conversion_tasks(megatron_model) # mcore→HF 的“全局转换目录”: 一个 mcore 参数怎么转换成哪些 HF tensor
0228:     index: list[McoreParamExport] = []

0230:         mapping = task.mapping # 该参数的 megatron→HF 映射器
0231:         param = task.param_weight # 本 rank 持有的 mcore 参数
0232:         name = task.global_param_name
0233:         if param is None:
0234:             if pp_world == 1:
0235:                 # not a pp placeholder (e.g. a skipped mapping); nothing to ship
0236:                 continue
0237:             # owned by another pipeline stage: zero-count lockstep row -- the
0238:             # rank walks the same global directory and contributes nothing.
0239:             index.append(
0240:                 McoreParamExport(
0241:                     megatron_name=name,
0242:                     param=None,
0243:                     spec=ShardSpec(full_shape=(0,), place=0, gather_group=world, contributes=False),
0244:                     probe=None,
0245:                     module=None,

0251: 
0252:         is_expert = mapping.is_expert and (ep_size > 1 or tp_world > 1)
0253:         is_tp_sharded = getattr(param, "tensor_model_parallel", False) and tp_world > 1
0254: 
0255:         if pp_world > 1:

0257:             # rank 0) sits in every gather regardless of which stage owns the

0312:         index = [rec for rec in index if rec.slots is not None]

0316: def _exchange_slot_tables(index: list[McoreParamExport], slot_cache: dict) -> None:
0317:     """One-time GLOBAL slot-table exchange, run at every world size, keyed by
0318:     DIRECTORY ROW (not by param name).
0319: 
0320:     The engine's batched gather merges entries BY SLOT POSITION within each
0321:     directory row and rank 0 names the merged pieces from ITS OWN list, so
0322:     the per-row list must be identical on every rank of the row's merge
0323:     group. Name-keyed merging is not enough on mcore: expert param NAMES
0324:     embed the expert ids (``...experts.linear_fc1.weight0`` on ep rank 0 vs
0325:     ``weight1`` on ep rank 1), so the same row carries different names per
0326:     rank while still gathering together. Each rank therefore probes a zero
0327:     delta through its OWNED rows to reveal its local ``(hf_name, hf_shape)``
0328:     list, one ``all_gather_object`` over WORLD exchanges the ordered row
0329:     lists, and every row's table becomes the rank-order first-seen union --
0330:     identical everywhere by construction. Dense/TP rows (same list on every
0331:     rank) dedup to themselves; expert rows concatenate the per-ep-rank
0332:     lists; PP placeholder rows inherit the owners' union."""
0333:     local_rows: list = []
0334:     for rec in index:
0335:         if rec.param is None:
0336:             local_rows.append(None)
0337:             continue
0338:         if rec.megatron_name not in slot_cache:
0339:             empty_idx = torch.empty(0, dtype=torch.int64, device=rec.param.device)
0340:             empty_val = torch.empty(0, dtype=torch.bfloat16, device=rec.param.device)
0341:             # 让 mcore_hf_delta_entry 在纯 NaN buffer 上跑真实转换路径，从而“揭示”本 rank 的本地 (hf_name, hf_shape) 列表并写进 slot_cache
0342:             mcore_hf_delta_entry(rec, 0, empty_idx, empty_val, slot_cache) # 喂一个零 delta 跑一次 probe
0343:         local_rows.append(slot_cache[rec.megatron_name])
0344: 
0345:     world = torch.distributed.get_world_size()
0346:     gathered: list = [None] * world
0347:     torch.distributed.all_gather_object(gathered, local_rows) # 获取全局的情况 -- 增加 和 去重
0348:     n_rows = len(local_rows)
0349:     assert all(len(rows) == n_rows for rows in gathered), ( # 所有 rank 的行数必须一致（bridge 的全局枚举是结构性并行的 没太懂
0350:         f"directory row counts diverge across ranks: {[len(r) for r in gathered]} -- "
0351:         "the bridge's global enumeration is expected to be structurally parallel"
0352:     )
0353:     for k, rec in enumerate(index):
0354:         union: dict = {}
0355:         for rows in gathered:  # rank order -> identical result on every rank
0356:             row = rows[k] # 各 rank 报自己那行会产出哪些 HF slot（零 delta probe）→ 一次对象级 all_gather → 按行拼成一张全组逐位一致的总表”
0357:             if row is None:
0358:                 continue
0359:             for slot in row:
0360:                 union[(slot[0], tuple(slot[1]))] = None  # ordered-set semantics
0361:         rec.slots = [(n, tuple(shape)) for (n, shape) in union.keys()] if union else None

0364: def mcore_hf_delta_entry(rec: McoreParamExport, _place, lidx: torch.Tensor, lval: torch.Tensor, slot_cache: dict):
0365:     """Probe one mcore param's shard-local delta into its final HF-coordinate
0366:     entry ``(slots, dtype_str, counts, hf_idx, hf_val)``.
0367: 
0368:     Scatters the delta into a NaN buffer of the LOCAL shard shape (exactly what
0369:     the real ``megatron_to_hf`` receives), runs the comm-stubbed probe -- real
0370:     group sizes, gathers synthesized locally, so the mapping executes its real
0371:     TP>1 code paths -- and extracts each output slot's surviving positions.
0372:     The slot list is cached after the first call (the converter's output names
0373:     are deterministic, so every rank's cache agrees and the batched gather
0374:     stays aligned)."""
0375:     dtype_str = str(lval.dtype).replace("torch.", "")
0376:     assert lval.numel() == 0 or lval.is_floating_point(), (
0377:         f"{rec.megatron_name}: NaN sentinels require a floating-point param, got {lval.dtype}"
0378:     )
0379: 
0380:     cached = rec.slots if rec.slots is not None else slot_cache.get(rec.megatron_name)
0381:     if rec.param is None:
0382:         # owned by another pipeline stage: pure lockstep row. The slot table
0383:         # was pre-seeded from the owner stage (fail loud if not -- an
0384:         # unsized entry would silently misalign the batched gather).
0385:         assert cached is not None, f"{rec.megatron_name}: slot table not pre-seeded for a non-owned PP param"
0386:         assert lidx.numel() == 0, f"{rec.megatron_name}: delta reported for a param this rank does not own"
0387:     if lidx.numel() == 0 and cached is not None: # delta 为空且表已有 → 直接返回零计数条目
0388:         # empty delta: the slot table froze after the first probe, so the
0389:         # zero-count lockstep entry needs no probe run at all -- skip the
0390:         # buffer build, the transform and the full-output NaN scan.
0391:         return (
0392:             cached,
0393:             dtype_str,
0394:             torch.zeros(len(cached), dtype=torch.int64),
0395:             torch.empty(0, dtype=torch.int32, device=lval.device),
0396:             torch.empty(0, dtype=lval.dtype, device=lval.device),
0397:         )
0398: 
0399:     buf = torch.full(tuple(rec.param.shape), float("nan"), dtype=lval.dtype, device=lval.device) # 初始化一份 NAN buf
0400:     if lidx.numel():
0401:         buf.view(-1)[lidx] = lval
0402: 
0403:     outs = rec.probe.megatron_to_hf(buf, rec.module)
0404: 
0405:     key = rec.megatron_name
0406:     slots = rec.slots if rec.slots is not None else slot_cache.get(key)
0407:     if slots is None:
0408:         # only reachable from the index-build exchange itself, which probes a
0409:         # zero delta to reveal this rank's local list before the row unions
0410:         # are installed on the recs; steady entries always find rec.slots.
0411:         slots = [(n, tuple(int(x) for x in t.shape)) for n, t in outs.items()]
0412:         slot_cache[key] = slots
0413:     unknown = set(outs) - {n for n, _ in slots}
0414:     assert not unknown, (
0415:         f"{key}: probe emitted slots missing from the global table {sorted(unknown)[:4]} -- "
0416:         "non-deterministic converter naming would misalign the batched gather"
0417:     )
0418:     counts = torch.zeros(len(slots), dtype=torch.int64)
0419:     idx_pieces: list[torch.Tensor] = []
0420:     val_pieces: list[torch.Tensor] = []
0421:     for s_i, (sname, _sshape) in enumerate(slots):
0422:         out = outs.get(sname)
0423:         if out is None:
0424:             continue  # another rank's slot (e.g. its ep-local experts): zero count here
0425:         fl = out.reshape(-1)
0426:         p_ = (~torch.isnan(fl)).nonzero(as_tuple=False).view(-1)
0427:         if p_.numel():
0428:             counts[s_i] = p_.numel()
