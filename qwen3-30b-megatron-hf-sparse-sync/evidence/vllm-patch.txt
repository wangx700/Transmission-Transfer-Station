0057:     # Enforce the sparse-loader contract and preserve destination values where
0058:     # the corresponding source is NaN.
0059:     def copy_non_nan_(
0060:         destination: torch.Tensor,
0061:         source: torch.Tensor,
0062:         non_blocking: bool = False,
0063:     ) -> torch.Tensor:
0064:         if not (
0065:             destination.is_floating_point()
0066:             and source.is_floating_point()
0067:             and destination.shape == source.shape
0068:         ):
0069:             raise NotImplementedError(
0070:                 "Sparse checkpoint patches require a same-shaped "
0071:                 "floating-point final copy"
0072:             )
0073:         source = source.to(dtype=destination.dtype, device=destination.device)
0074:         return torch.where(
0075:             torch.isnan(source),
0076:             destination,
0077:             source,
0078:             out=destination,
0079:         )

0083:         return model.load_weights(weights)
0084:     finally:
0085:         torch.Tensor.copy_ = original_copy

0088: def _validate_patch_structure(patch: CheckpointWeightPatch) -> int:
0089:     if not patch.name:
0090:         raise ValueError("Checkpoint weight name must be non-empty")
0091:     if not isinstance(patch.dtype, torch.dtype):
0092:         raise TypeError(f"{patch.name}: dtype must be a torch.dtype")
0093:     if any(dim < 0 for dim in patch.shape):
0094:         raise ValueError(f"{patch.name}: shape dimensions must be non-negative")
0095:     if patch.values.ndim != 1:
0096:         raise ValueError(f"{patch.name}: patch values must be a 1D tensor")
0097: 
0098:     numel = math.prod(patch.shape)
0099:     if patch.indices is None:
0100:         if patch.values.numel() != numel:
0101:             raise ValueError(
0102:                 f"{patch.name}: dense patch has {patch.values.numel()} values "
0103:                 f"for a {numel}-element tensor"
0104:             )
0105:         return numel
0106: 
0107:     if not patch.dtype.is_floating_point:
0108:         raise TypeError(
0109:             f"{patch.name}: sparse checkpoint patches require a floating dtype "
0110:             "because NaN is the unchanged-value sentinel"
0111:         )
0112:     if patch.indices.ndim != 1:
0113:         raise ValueError(f"{patch.name}: sparse patch indices must be a 1D tensor")
0114:     if patch.indices.dtype not in (torch.int32, torch.int64):
0115:         raise TypeError(f"{patch.name}: sparse patch indices must be int32 or int64")
0116:     if patch.indices.numel() != patch.values.numel():
0117:         raise ValueError(
0118:             f"{patch.name}: sparse indices and values must have matching lengths"
0119:         )
0120: 
0121:     return numel

0140: def load_checkpoint_weight_patches(
0141:     model: torch.nn.Module,
0142:     patches: Iterable[CheckpointWeightPatch],

0145:     validate_unique_indices: bool = True,

0147:     sparse_apply=None,

0150:     """Load ordered patches through the model's checkpoint loader.
0151: 
0152:     A call may contain either dense or sparse patches, but not both. A repeated
0153:     name starts a new loader call, so patches for that weight are applied in input
0154:     order. A later patch may update positions changed by an earlier patch. Each
0155:     sparse patch creates a full checkpoint-shaped tensor whose NaNs mark unchanged
0156:     elements. Each locally applied sparse patch must use one final same-shaped
0157:     floating-point ``Tensor.copy_``. Intermediate or unrelated ``copy_`` calls,
0158:     composed loaders, and custom write paths are unsupported. ``max_chunk_bytes``
0159:     is a batching target; one tensor may exceed it.
0160: 
0161:     Online checkpoint-format dense updates require the caller to manage vLLM's
0162:     layerwise reload lifecycle. Sparse online updates modify initialized model
0163:     tensors in place and must not use that lifecycle. Patch shapes, dtypes, value
0164:     lengths, and sparse indices are validated before loading begins.
0165: 
0166:     This function does not roll back a partial update. If a loader fails after a
0167:     write, some destinations may already be changed. Do not serve from the affected
0168:     worker until a known baseline has been restored or the worker has been
0169:     restarted.
0170: 
0171:     Args:
0172:         model: Model whose native ``load_weights`` method applies the patches.
0173:         patches: Dense replacements or sparse updates in checkpoint coordinates.
0174:         max_chunk_bytes: Target checkpoint tensor bytes per
0175:             ``model.load_weights`` call. A single tensor may exceed this target.
0176:         validate_unique_indices: Whether to reject duplicate sparse indices within
0177:             each patch. Repeated patches may update the same positions. Disable
0178:             only for a trusted producer that already guarantees unique positions
0179:             within each patch.
0180:         sparse_apply: Optional extension called after validation for sparse
0181:             updates. Returns (remaining patches in original order, handled weight
0182:             names). It must preserve ordered replacement semantics and may mutate
0183:             model weights. Failure is nontransactional, as with native loading.
0184:         load_context: Optional context-manager factory entered only around native
0185:             model loading, after the sparse extension. Useful for loader views.
0186:         phase_observer: Optional context-manager factory receiving a phase name
0187:             for diagnostic timing. No device synchronization is added by default.
0188: 
0189:     Returns:
0190:         Union of the weight names reported by all ``model.load_weights`` calls.
0191:     """
